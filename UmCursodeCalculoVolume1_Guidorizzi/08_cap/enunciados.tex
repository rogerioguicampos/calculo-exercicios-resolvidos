\vspace{0.5cm}
\noindent\textbf{Exercício 1.} Calcule.
\begin{enumerate}
    \renewcommand{\theenumi}{\alph{enumi}}
    \renewcommand{\labelenumi}{\theenumi)}
    \item $\operatorname{arcsen} 1$
    \item $\operatorname{arcsen} \dfrac{1}{2}$
    \item $\operatorname{arcsen} \dfrac{\sqrt{3}}{2}$
    \item $\operatorname{arctg} 1$
    \item $\operatorname{arctg} (-1)$
    \item $\operatorname{arctg} \sqrt{3}$
    \item $\operatorname{arcsen} \left(-\dfrac{1}{2}\right)$
    \item $\operatorname{arcsen} (-1)$
    \item $\operatorname{arctg} (-\sqrt{3})$
    \item $\operatorname{arcsen} \left(-\dfrac{\sqrt{3}}{2}\right)$
    \item $\operatorname{arctg} \dfrac{\sqrt{3}}{3}$
    \item $\operatorname{arctg} \left(-\dfrac{\sqrt{3}}{3}\right)$
\end{enumerate}

\vspace{0.5cm}
\noindent\textbf{Exercício 2.} Verifique que
\begin{enumerate}
    \renewcommand{\theenumi}{\alph{enumi}}
    \renewcommand{\labelenumi}{\theenumi)}
    \item $\cos (\operatorname{arcsen} x) = \sqrt{1 - x^2}$
    \item $\sec (\operatorname{arctg} x) = \sqrt{1 + x^2}$
\end{enumerate}

\vspace{0.5cm}
\noindent\textbf{Exercício 3.} Calcule.
\begin{enumerate}
    \renewcommand{\theenumi}{\alph{enumi}}
    \renewcommand{\labelenumi}{\theenumi)}
    \item $\cos \left(\operatorname{arcsen} \dfrac{1}{2}\right)$
    \item $\cos \left(\operatorname{arcsen} \dfrac{\sqrt{3}}{2}\right)$
    \item $\cos \left(\operatorname{arcsen} \left(-\dfrac{\sqrt{3}}{2}\right)\right)$
    \item $\sec (\operatorname{arctg} 1)$
    \item $\operatorname{arcsen} (\operatorname{sen} x)$, em que $-\dfrac{\pi}{2} \le x \le \dfrac{\pi}{2}$
    \item $\operatorname{arctg} (\operatorname{tg} x)$, em que $-\dfrac{\pi}{2} < x < \dfrac{\pi}{2}$
    \item $\operatorname{arcsen} \left(\operatorname{sen} \dfrac{2\pi}{3}\right)$ (Cuidado!)
    \item $\operatorname{arcsen} (\operatorname{sen} 3\pi)$
    \item $\operatorname{arcsen} \left(\operatorname{sen} \dfrac{5\pi}{3}\right)$
    \item $\operatorname{arcsen} (\operatorname{sen} x)$, em que $x = 2k\pi + \bar{x}$, $k$ inteiro e $\bar{x} \in \left[-\dfrac{\pi}{2}, \dfrac{\pi}{2}\right]$
\end{enumerate}

\vspace{0.5cm}
\noindent\textbf{Exercício 4.} Seja $f$ uma função inversível com inversa $g$. Mostre que
\begin{enumerate}
    \renewcommand{\theenumi}{\alph{enumi}}
    \renewcommand{\labelenumi}{\theenumi)}
    \item $f(g(x)) = x$ para todo $x \in D_g$
    \item $g(f(x)) = x$ para todo $x \in D_f$
\end{enumerate}

\subsection{Derivada de Função Inversa}
\vspace{0.5cm}
\noindent\textbf{Exercício 1.} Determine a derivada.
\begin{enumerate}
    \renewcommand{\theenumi}{\alph{enumi}}
    \renewcommand{\labelenumi}{\theenumi)}
    \item $y = x\operatorname{arctg} x$
    \item $f(x) = \operatorname{arcsen} 3x$
    \item $g(x) = \operatorname{arcsen} x^3$
    \item $y = \operatorname{arctg} x^2$
    \item $y = 3\operatorname{arctg}(2x + 3)$
    \item $y = \operatorname{arcsen} e^x$
    \item $y = e^{3x}\operatorname{arcsen} 2x$
    \item $y = \dfrac{\operatorname{sen} 3x}{\operatorname{arctg} 4x}$
    \item $y = x^2 e^{\operatorname{arctg} 2x}$
    \item $y = \dfrac{x\operatorname{arctg} x}{\cos 2x}$
    \item $y = e^{-3x} + \ln(\operatorname{arctg} x)$
    \item $f(x) = \dfrac{e^{-x}\operatorname{arctg} e^x}{\operatorname{tg} x}$
\end{enumerate}

\vspace{0.5cm}
\noindent\textbf{Exercício 2.}
Seja $f(x) = x + e^x$ e seja $g$ a inversa de $f$. Mostre que $g$ é derivável e que $g'(x) = \dfrac{1}{1 + e^{g(x)}}$. (Sugestão: Veja Exercício 13-8.1.)

\vspace{0.5cm}
\noindent\textbf{Exercício 3.}
Seja $f(x) = x + e^x$ e seja $g$ a função inversa de $f$. Calcule $g'(1)$ e $g''(1)$.

\vspace{0.5cm}
\noindent\textbf{Exercício 4.} Seja $f(x) = x + \ln x$, $x > 0$.
\begin{enumerate}
    \renewcommand{\theenumi}{\alph{enumi}}
    \renewcommand{\labelenumi}{\theenumi)}
    \item Mostre que $f$ admite função inversa $g$, que $g$ é derivável e que $g'(x) = \dfrac{g(x)}{1 + g(x)}$
    \item Esboce os gráficos de $f$ e de $g$
    \item Calcule $g(1)$, $g'(1)$ e $g''(1)$
\end{enumerate}

\vspace{0.5cm}
\noindent\textbf{Exercício 5.} Seja $f(x) = x + x^3$.

\noindent\textbf{Exercício 7.}
(Função arco-secante.) A função $f(x) = \sec x$, $0 \le x < \dfrac{\pi}{2}$ é inversível e sua inversa é a função $g(x) = \operatorname{arcsec} x$, $x \ge 1$. Calcule $\operatorname{arcsec}' x$.

\vspace{0.5cm}
\noindent\textbf{Exercício 8.} Verifique que
\begin{enumerate}
    \renewcommand{\theenumi}{\alph{enumi}}
    \renewcommand{\labelenumi}{\theenumi)}
    \item $\dfrac{d}{dx}\!\left[x\operatorname{arctg} x - \dfrac{1}{2}\ln(1 + x^2)\right] = \operatorname{arctg} x$
    \item $\dfrac{d}{dx}\!\left[\dfrac{x^3}{3}\operatorname{arcsen} x + \dfrac{x^2 + 2}{9}\sqrt{1 - x^2}\right] = x^2 \operatorname{arcsen} x$
    \item $\dfrac{d}{dx}\left[(x + 1)\operatorname{arctg}\sqrt{x} - \sqrt{x}\right] = \operatorname{arctg}\sqrt{x}$
    \item $\dfrac{d}{dx}\left[\dfrac{1}{2}\operatorname{arcsen}\left(\dfrac{2 - x}{x\sqrt{2}}\right)\right] = -\dfrac{1}{x\sqrt{x^2 + 4x - 4}}$
    \item $\dfrac{d}{dx}\left[\dfrac{\sqrt{27x^2 + 6x - 1}}{x} - 3\operatorname{arcsen}\left(\dfrac{1 - 3x}{6x}\right)\right] = \dfrac{1}{x^2 \sqrt{27x^2 + 6x - 1}}$
    \item $\dfrac{d}{dx}\left[\sqrt{\dfrac{2}{3}}\operatorname{arctg}\sqrt{\dfrac{2(3 - x)}{3(x - 2)}}\right] = -\dfrac{1}{x\sqrt{5x - 6 - x^2}}$
    \item $\dfrac{d}{dx}\left[\dfrac{1}{36}x(9x^2 - 2)\sqrt{4 - 9x^2} + \dfrac{2}{27}\operatorname{arcsen}\dfrac{3x}{2}\right] = x^2 \sqrt{4 - 9x^2}$
\end{enumerate}
