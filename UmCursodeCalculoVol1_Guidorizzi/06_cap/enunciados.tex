% ==============================================================================
% 6.1. POTÊNCIA COM EXPOENTE REAL
% ==============================================================================
\subsection{Potência com Expoente Real}

\begin{exercicio}[1]
Calcule:
\begin{enumerate}[label=\alph*)]
    \item $\lim\limits_{x \to +\infty} 3^x$
    \item $\lim\limits_{x \to -\infty} 5^x$
    \item $\lim\limits_{x \to -\infty} e^x$
    \item $\lim\limits_{x \to +\infty} (0{,}13)^x$
    \item $\lim\limits_{x \to +\infty} [2^x - 3^x]$
    \item $\lim\limits_{x \to +\infty} \dfrac{1 - 2^x}{1 - 3^x}$
    \item $\lim\limits_{x \to +\infty} 2^{-x}$
    \item $\lim\limits_{x \to +\infty} [2^x + 2^{-x}]$
    \item $\lim\limits_{x \to -\infty} 2^{-x}$
    \item $\lim\limits_{x \to -\infty} [2^x + 2^{-x}]$
\end{enumerate}
\end{exercicio}

\begin{exercicio}[2]
Esboce o gráfico:
\begin{enumerate}[label=\alph*)]
    \item $f(x) = 3^x$
    \item $g(x) = (0{,}12)^x$
    \item $f(x) = e^{-x}$
    \item $g(x) = 1 + e^{-x}$
    \item $f(x) = -e^{-x}$
    \item $g(x) = 1 - e^{-x}$
    \item $f(x) = e^x + e^{-x}$
    \item $g(x) = e^{-x}\sen x$
    \item $f(x) = e^{1/x}$
    \item $g(x) = e^{-x^2}$
\end{enumerate}
\end{exercicio}

% ==============================================================================
% 6.2. LOGARITMO
% ==============================================================================
\subsection{Logaritmo}

\begin{exercicio}[1]
Calcule:
\begin{enumerate}[label=\alph*)]
    \item $\log_{10} 100$
    \item $\log_{\frac{1}{2}} 16$
    \item $\log_{\frac{1}{2}} \sqrt{2}$
    \item $\log_9 \sqrt{3}$
    \item $\log_{10} 1$
    \item $\log_5 (-5)$
    \item $\log_a 1$ ($a > 0$ e $a \neq 1$)
    \item $\log_3 243$
\end{enumerate}
\end{exercicio}

\begin{exercicio}[2]
Determine o domínio:
\begin{enumerate}[label=\alph*)]
    \item $f(x) = \log_2 (x + 1)$
    \item $g(x) = \ln(x^2 - 1)$
    \item $g(x) = \ln(-x)$
    \item $f(x) = \log_3 |x|$
    \item $f(x) = \ln\dfrac{x + 1}{x - 1}$
    \item $g(x) = \log_x 3$
\end{enumerate}
\end{exercicio}

\begin{exercicio}[3]
Ache o domínio e esboce o gráfico:
\begin{enumerate}[label=\alph*)]
    \item $f(x) = \log_3 x$
    \item $g(x) = \ln x$
    \item $f(x) = \log_{\frac{1}{3}} x$
    \item $g(x) = \ln(x - 1)$
    \item $f(x) = \ln(-x)$
    \item $g(x) = \ln |x|$
    \item $f(x) = |\ln x|$
    \item $g(x) = |\ln |x||$
\end{enumerate}
\end{exercicio}

\begin{exercicio}[4]
Calcule:
\begin{enumerate}[label=\alph*)]
    \item $\lim\limits_{x \to +\infty} \log_3 x$
    \item $\lim\limits_{x \to 0^+} \log_{\frac{1}{3}} x$
    \item $\lim\limits_{x \to 0^+} \ln x$
    \item $\lim\limits_{x \to +\infty} \ln\dfrac{x}{x + 1}$
    \item $\lim\limits_{x \to +\infty} [\ln(2x + 1) - \ln(x + 3)]$
    \item $\lim\limits_{x \to 1} \ln\dfrac{x^2 - 1}{x - 1}$
    \item $\lim\limits_{x \to +\infty} [x\ln 2 - \ln(3^x + 1)]$
\end{enumerate}
\end{exercicio}

% ==============================================================================
% 6.3. O LIMITE \lim_{x \to +\infty} (1 + 1/x)^x
% ==============================================================================
\subsection{O Limite $\lim\limits_{x \to +\infty} \left(1 + \frac{1}{x}\right)^x$}

\begin{exercicio}[1]
Calcule:
\begin{enumerate}[label=\alph*)]
    \item $\lim\limits_{x \to +\infty} \left(1 + \dfrac{2}{x}\right)^{\!x}$
    \item $\lim\limits_{x \to +\infty} \left(1 + \dfrac{1}{x}\right)^{\!x+2}$
    \item $\lim\limits_{x \to +\infty} \left(1 + \dfrac{1}{2x}\right)^{\!x}$
    \item $\lim\limits_{x \to +\infty} \left(1 + \dfrac{2}{x}\right)^{\!x+1}$
    \item $\lim\limits_{x \to +\infty} \left(\dfrac{x + 2}{x + 1}\right)^{\!x}$
    \item $\lim\limits_{x \to 0} (1 + 2x)^x$
    \item $\lim\limits_{x \to 0} (1 + 2x)^{\frac{1}{x}}$
    \item $\lim\limits_{x \to +\infty} \left(1 + \dfrac{1}{x}\right)^{\!2x}$
\end{enumerate}
\end{exercicio}

\begin{exercicio}[2]
Seja $a > 0$, $a \neq 1$. Mostre que:
\[
\lim_{h \to 0} \dfrac{a^h - 1}{h} = \ln a.
\]
\end{exercicio}

\begin{exercicio}[3]
Calcule:
\begin{enumerate}[label=\alph*)]
    \item $\lim\limits_{x \to 0} \dfrac{e^{2x} - 1}{x}$
    \item $\lim\limits_{x \to 0} \dfrac{e^{x^2} - 1}{x}$
    \item $\lim\limits_{x \to 0} \dfrac{5^x - 1}{x}$
    \item $\lim\limits_{x \to 0^+} \dfrac{3^x - 1}{x^2}$
\end{enumerate}
\end{exercicio}
