\newcommand{\defderivadaformal}[1][]{%
    \begin{teoriabox}[title=Definição Formal de Derivada num Ponto, #1]
        Sejam $f$ uma função e $p$ um ponto de seu domínio. O limite
        \[
        \lim_{x \to p} \frac{f(x) - f(p)}{x - p}
        \]
        quando existe e é finito, denomina-se \textbf{derivada de $f$ em $p$} e indica-se por $f'(p)$. Se $f$ admite derivada em $p$, dizemos que $f$ é \textbf{derivável} (ou diferenciável) em $p$.
    \end{teoriabox}%
}

\newcommand{\obsdefderivada}[1][]{%
    \begin{observacaoteoriabox}[title=Equivalência das Formas de Cálculo, #1]
        A definição por aproximação pontual e a definição por acréscimo são equivalentes mediante a mudança de variável $x = p + h$ (onde $x \to p \iff h \to 0$):
        \[
        f'(p) = \lim_{x \to p} \frac{f(x) - f(p)}{x - p} = \lim_{h \to 0} \frac{f(p + h) - f(p)}{h}
        \]
        \textbf{Critério de Escolha Operacional:}
        \begin{itemize}[noitemsep, topsep=2pt]
            \item \textbf{Forma pontual ($x \to p$):} Indicada quando $p$ é um número fixo, pois simplifica o cancelamento do fator $(x - p)$ por fatoração direta ou conjugados.
            \item \textbf{Forma por acréscimo ($h \to 0$):} Indicada para determinar a função derivada geral $f'(x)$, expandindo binômios e simplificando o fator comum $h$.
        \end{itemize}
    \end{observacaoteoriabox}%
}

\newcommand{\defretatangente}[1][]{%
  \begin{teoriabox}[title=Reta Tangente ao Gráfico de $f$ no ponto $(p, f(p))$, #1]
        \noindent
        \begin{minipage}[c]{0.56\textwidth}
            A derivada de $f$, em $p$, é o coeficiente angular da reta tangente ao gráfico de $f$ no ponto de abscissa $p$.
            
            \vspace{0.2cm}
            A equação da reta tangente ao gráfico de $f$ em $(p, f(p))$ é dada por:
            \[
            y - f(p) = f'(p)(x - p)
            \]
        \end{minipage}\hfill
        \begin{minipage}[c]{0.40\textwidth}
            \centering
            \begin{tikzpicture}[>=latex, scale=0.9]
                % Eixos cartesianos
                \draw[->, thick, gray!70] (-0.3, 0) -- (3.3, 0) node[below, black] {\scriptsize $x$};
                \draw[->, thick, gray!70] (0, -0.3) -- (0, 2.7) node[left, black] {\scriptsize $y$};
                
                % Ponto de Tangência P = (p, f(p)) em x = 1.45
                \coordinate (P) at (1.45, 0.95);
                
                % Curva f(x)
                \draw[thick, teal!70!black, smooth, domain=0.5:2.65] 
                    plot (\x, {0.35*(\x)^2 - 0.2*\x + 0.5});
                \node[above, teal!70!black] at (2.4, 2.25) {\scriptsize $y = f(x)$};
                
                % Reta Tangente
                \draw[line width=1pt, blue!70!black, domain=0.25:2.55] 
                    plot (\x, {0.815*(\x - 1.45) + 0.95});
                \node[right, blue!70!black] at (2.4, 1.45) {\scriptsize $T$};
                
                % Projeções pontilhadas nos eixos
                \draw[densely dashed, gray!60] (1.45, 0) node[below, black] {\scriptsize $p$} -- (P);
                \draw[densely dashed, gray!60] (0, 0.95) node[left, black] {\scriptsize $f(p)$} -- (P);
                
                % Ponto de tangência
                \fill[teal!70!black] (P) circle (1.3pt);
                
                % Rótulo do ponto: posicionado acima da curva à esquerda, sem tocar o eixo y
                \node[above, xshift=-10pt, yshift=2pt, teal!80!black] at (P) {\scriptsize $(p, f(p))$};
            \end{tikzpicture}
        \end{minipage}
    \end{teoriabox}%
}
